    d) 5

18.
    def mcd(m, n):
        menor = min(m, n)
        mayor = max(m, n)
        resto = mayor % menor
        if resto == 0:
            return menor
        else:
            return mcd(menor, resto)
    
    print(mcd(12, 18))

    ¿Qué se imprime?
    a) 12
    b) 36
    c) 6
    d) 18

19.
    def integral_x2(a, b, n):
        deltax = (b - a) / n
        sumatorio = 0.0
        for i in range(n):
            sumatorio += deltax * (a + i * deltax) ** 2
        return sumatorio
    
    print(integral_x2(0, 2, 2))

    ¿Qué se imprime?
    a) 0.5
    b) 2.0
    c) 0.0
    d) 1.0

    def pot(b, e):
        if e == 0:
            return 1
        return b * pot(b, e - 1)
    
    print(pot(2, 5))

    ¿Qué se imprime?
    a) 25
    b) 32
    c) 16
    d) 10
